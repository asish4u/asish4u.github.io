from pyspark.sql import SparkSession
from delta import DeltaTable

def build_pipeline(config):
    spark = SparkSession.builder \
        .appName("EHR_Claims_ETL") \
        .config("spark.databricks.delta", "true") \
        .getOrCreate()
    
    # Extract from source systems
    raw_claims = spark.read \
        .format("jdbc") \
        .option("url", config.jdbc_url) \
        .option("dbtable", "claims") \
        .load()
    
    # Transform with business logic
    enriched = raw_claims \
        .withColumn("claim_category", categorize_udf("diagnosis_code")) \
        .withColumn("risk_score", calculate_risk("member_id")) \
        .filter("claim_status = 'PAID'")
    
    # Load to Snowflake via Delta Lake
    enriched.write \
        .format("snowflake") \
        .option("dbtable", "FACT_CLAIMS") \
        .mode("overwrite") \
        .save()
    
    return "Pipeline completed successfully"
